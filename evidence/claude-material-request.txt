const keys = keyboardLayout(21, 108);
const KEY_WIDTH = 42;
const WIDTH = 52 * KEY_WIDTH;
const positions = new Map(keys.map((k) => [k.midi, k]));

export default function KeyboardGuide({

  active,
  soundEvents = [],
  pressed,
  tonic,
  labels,
  onNoteOn,
  onNoteOff,
  onRelease,

  const clearPointer = (id) =>
    setPointerHolds((previous) => {
      const next = new Set(previous);
      next.delete(id);
      return next;
    });

  const pitches = targets.map((n) => n.midi);
  const soundTokens = new Map();
  for (const note of soundEvents) {
    soundTokens.set(
      note.midi,
      `${soundTokens.get(note.midi) || ""}:${note.id ?? note.beat}`,
    );
  }
  const targetKey = pitches.join(",");
  const targetPositions = pitches
    .map((m) => positions.get(m))
    .filter(Boolean)
    .sort((a, b) => a.left - b.left);
  const low = targetPositions.length
    ? Math.min(
        ...targetPositions.map((k) => ((k.left + k.width / 2) / 100) * WIDTH),
      )
    : WIDTH / 2;
  const high = targetPositions.length
    ? Math.max(
        ...targetPositions.map((k) => ((k.left + k.width / 2) / 100) * WIDTH),
      )
    : low;

    labels === "letters"
      ? noteName(m)
      : labels === "solfege"
        ? solfege(m, tonic)
        : degree(m, tonic);

      low < el.scrollLeft + KEY_WIDTH ||
      high > el.scrollLeft + el.clientWidth - KEY_WIDTH

      el.scrollLeft = Math.max(
        0,
        Math.min(WIDTH - el.clientWidth, (low + high - el.clientWidth) / 2),
      );
      readWindow();

    window.addEventListener("blur", clear);
    return () => window.removeEventListener("blur", clear);

      readWindow();
    });
    observer.observe(viewport.current);
    readWindow();
    return () => observer.disconnect();
  }, []);
  useEffect(() => {
    setFollow(true);
    center(true);
  }, [scoreId]);

              key={k.midi}
              className={`${k.black ? "black" : "white"} ${pitches.includes(k.midi) ? "target" : ""}`}

              width: (viewportState.width / WIDTH) * 100 + "%",
            }}
          />
        </div>
        <span>C8</span>

                tabIndex={k.midi === focusPitch ? 0 : -1}
                onFocus={() => setFocusPitch(k.midi)}
                aria-label={`${noteName(k.midi)} ${degree(k.midi, tonic)} ${solfege(k.midi, tonic)}`}
                className={`piano-key ${k.black ? "black" : "white"} ${active.has(k.midi) || pitches.includes(k.midi) ? "lit" : ""} ${pressed.has(k.midi) ? "pressed" : ""} ${pressed.has(k.midi) || soundTokens.has(k.midi) ? "sounding" : ""} ${k.midi === 60 ? "middle-c" : ""}`}
                style={{ left: k.left + "%", width: k.width + "%" }}
                onPointerDown={(e) => {
                  e.preventDefault();
                  e.currentTarget.setPointerCapture(e.pointerId);
                  setPointerHolds((previous) =>
                    new Set(previous).add(e.pointerId),
                  );
                  onNoteOn(k.midi, 0.75, "pointer:" + e.pointerId);
                }}
                onPointerUp={(e) => {
                  clearPointer(e.pointerId);
                  onNoteOff(k.midi, "pointer:" + e.pointerId);
                }}
                onPointerCancel={(e) => {
                  clearPointer(e.pointerId);
                  onNoteOff(k.midi, "pointer:" + e.pointerId);
                }}
                onLostPointerCapture={(e) => {
                  clearPointer(e.pointerId);
                  onNoteOff(k.midi, "pointer:" + e.pointerId);
                }}
                onKeyDown={(e) => {
                  if (
                    ["ArrowLeft", "ArrowRight", "Home", "End"].includes(e.key)
                  ) {
                    e.preventDefault();
                    e.stopPropagation();
                    setFollow(false);
                    const pitch =
                      e.key === "Home"
                        ? 21
                        : e.key === "End"
                          ? 108
                          : Math.max(
                              21,
                              Math.min(
                                108,
                                k.midi + (e.key === "ArrowRight" ? 1 : -1),
                              ),
                            );
                    viewport.current
                      .querySelector(`[data-midi="${pitch}"]`)
                      ?.focus();
                    return;
                  }
                  if ((e.key === "Enter" || e.code === "Space") && !e.repeat) {
                    e.preventDefault();
                    onNoteOn(
                      k.midi,
                      0.75,
                      `key-button:${k.midi}:${e.code || e.key}`,
                    );
                  }
                }}
                onKeyUp={(e) => {
                  if (e.key === "Enter" || e.code === "Space") {
                    e.preventDefault();
                    onNoteOff(
                      k.midi,
                      `key-button:${k.midi}:${e.code || e.key}`,
                    );
                  }
                }}
                onBlur={() => onRelease(`key-button:${k.midi}:`)}
              >
                <span className="key-surface" aria-hidden="true">
                  <span
                    className="key-resonance"
                    key={`${pressed.has(k.midi) ? "live" : ""}${soundTokens.get(k.midi) || ""}`}
                  />
                  <span className="key-number">{label(k.midi)}</span>
                  <small>{noteName(k.midi)}</small>
                </span>
              </button>
            ))}
          </div>
        </div>
      </div>
      <div className="keyboard-guide-caption">
        <span>
          1 = Do / {noteName(60 + tonic).replace("4", "")} · numbers are notes,

        </span>
        <span>Scroll to explore all 88 keys</span>
      </div>
    </section>
  );

import { useEffect, useMemo, useRef, useState } from "react";
import { ChevronLeft, ChevronRight } from "lucide-react";
import { barStart, degree, noteName, solfege } from "./music.mjs";
import {
  indexReadingNotes,
  readingPassage,
  readingNotesBetween,
  readingGroups,
  readingRests,
  readingBars,
  readingDuration,
  readingSpacing,
  readingNoteDisplay,
} from "./reading.mjs";
import "./reading.css";

export default function ReadingScore({
  score,
  notes = score.notes,
  beat = 0,
  tonic = 0,
  labels = "numbers",
  hand = "both",
  onSeek,
  onStep,
}) {
  const root = useRef(null);
  const scroller = useRef(null);
  const sheet = useRef(null);
  const previousPassage = useRef(null);
  const [width, setWidth] = useState(960);
  const [jump, setJump] = useState("");
  const index = useMemo(() => indexReadingNotes(notes), [notes]);
  const barsPerPage = width < 760 ? 2 : 4;
  const passage = readingPassage(score, index, beat, barsPerPage);
  const { start, end } = passage;
  const content = useMemo(() => {
    const hands = ["right", "left"]
      .filter((name) => hand === "both" || hand === name)
      .map((hand) => {
        const visible = readingNotesBetween(index, hand, start, end);
        return {
          hand,
          notes: visible,
          groups: readingGroups(visible, start, end),
          rests: readingRests(visible, start, end),
        };
      });
    const onsets = [
      ...new Set(hands.flatMap((h) => h.groups.map((g) => g.beat))),
    ].sort((a, b) => a - b);
    const spacing = readingSpacing(start, end, onsets);
    return { hands, spacing, bars: readingBars(score, passage) };
  }, [index, score, start, end, hand]);
  const percent = (at) => `${content.spacing.position(at) * 100}%`;
  const widthPercent = (from, to) =>
    `${(content.spacing.position(to) - content.spacing.position(from)) * 100}%`;
  const seek = (at, direction = 0) => {
    if (onSeek) onSeek(Math.max(0, at));
    else onStep?.(direction, Math.max(0, at));
  };
  useEffect(() => {
    const observer = new ResizeObserver(([entry]) =>
      setWidth(entry.contentRect.width),
    );
    if (root.current) observer.observe(root.current);
    return () => observer.disconnect();
  }, []);
  useEffect(() => {
    if (scroller.current) scroller.current.scrollLeft = 0;
  }, [start, end]);
  useEffect(() => {
    const previous = previousPassage.current;
    previousPassage.current = { score: score.id, start };
    const motion = window.matchMedia("(prefers-reduced-motion: reduce)");
    if (motion.matches || !sheet.current?.animate) return;
    const direction =
      previous?.score === score.id && start < previous.start ? -1 : 1;
    // Animate the paper contents only. Keep the controls, focus and musical
    // cursor timing intact; this runs once per passage, never per audio frame.
    const entrance = sheet.current.animate(
      [
        { opacity: 0.64, transform: `translateX(${direction * 8}px)` },
        { opacity: 1, transform: "translateX(0)" },
      ],
      { duration: 260, easing: "cubic-bezier(.2,.7,.2,1)" },
    );
    const stopMotion = () => {
      if (motion.matches) entrance.cancel();
    };
    motion.addEventListener("change", stopMotion);
    return () => {
      entrance.cancel();
      motion.removeEventListener("change", stopMotion);
    };
  }, [score.id, start, end]);
  // On a dense page follow the reading position horizontally, without moving
  // the whole application or fighting manual scroll while the cursor is visible.
  useEffect(() => {
    const el = scroller.current;
    if (!el || el.scrollWidth <= el.clientWidth + 2) return;
    const x = content.spacing.position(beat) * (el.scrollWidth - 86) + 64;
    if (x > el.scrollLeft + el.clientWidth - 70 || x < el.scrollLeft + 64)
      el.scrollLeft = Math.max(0, x - el.clientWidth * 0.3);
  }, [beat, content.spacing]);
  const pageLabel =
    passage.firstBar === passage.lastBar
      ? `Bar ${passage.firstBar}`
      : `Bars ${passage.firstBar}–${passage.lastBar}`;
  const label = (m) =>
    labels === "letters"
      ? noteName(m).replace(/-?\d+$/, "")
      : labels === "solfege"
        ? solfege(m, tonic)
        : degree(m, tonic);
  return (
    <section
      className="reading-score"
      ref={root}
      aria-label="Sight-reading score"
      data-labels={labels}
    >
      <header className="reading-header">
        <div className="reading-heading">
          <span className="reading-eyebrow">READ FROM LEFT TO RIGHT</span>
          <h2>
            {pageLabel}
            <span> / {score.measures}</span>
          </h2>
        </div>
        <div className="reading-navigation">
          <form
            onSubmit={(e) => {
              e.preventDefault();
              const bar = Number(jump);
              if (Number.isInteger(bar) && bar >= 1 && bar <= score.measures) {
                seek(barStart(score, bar));
                setJump("");
              }
            }}
          >
            <label htmlFor="reading-bar-jump">Go to bar</label>
            <input
              id="reading-bar-jump"
              aria-label="Go to bar"
              type="number"
              min="1"
              max={score.measures}
              placeholder="—"
              value={jump}
              onChange={(e) => setJump(e.target.value)}
            />
            <button type="submit" className="reading-jump" disabled={!jump}>
              Go
            </button>
          </form>
          <button
            type="button"
            aria-label="Previous passage"
            title="Previous passage"
            disabled={passage.previous == null}
            onClick={() => {
              const previous = readingPassage(
                score,
                index,
                passage.previous,
                barsPerPage,
              );
              seek(previous.start, -1);
            }}
          >
            <ChevronLeft size={18} />
          </button>
          <button
            type="button"
            aria-label="Next passage"
            title="Next passage"
            disabled={passage.next == null}
            onClick={() => seek(passage.next, 1)}
          >
            <ChevronRight size={18} />
          </button>
        </div>
      </header>
      <div
        className="reading-scroll"
        ref={scroller}
        tabIndex="0"
        aria-label="Numbered score passage; scroll horizontally for dense music"
      >
        <div
          className="reading-sheet"
          ref={sheet}
          style={{ minWidth: content.spacing.width + 86 }}
        >
          <div className="reading-ruler">
            <span className="reading-gutter">HAND</span>
            <div className="reading-time">
              {content.bars.map((bar) => (
                <button
                  key={bar.bar}
                  type="button"
                  className="reading-bar-label"
                  style={{
                    left: percent(bar.start),
                    width: widthPercent(bar.start, bar.end),
                  }}
                  onClick={() => seek(barStart(score, bar.bar))}
                  aria-label={`Seek to bar ${bar.bar}`}
                >
                  {String(bar.bar).padStart(2, "0")}
                  {bar.continuation ? " · continued" : ""}
                </button>
              ))}
            </div>
          </div>
          {content.hands.map(({ hand, groups, rests }) => {
            const depth = Math.max(1, ...groups.map((g) => g.notes.length));
            return (
              <div
                className={`reading-hand reading-${hand} ${groups.length ? "" : "reading-empty"}`}
                key={hand}
              >
                <div className="reading-gutter reading-hand-name">
                  <strong>{hand === "right" ? "R" : "L"}</strong>
                  <span>{hand} hand</span>
                </div>
                <div
                  className="reading-lane"
                  style={{
                    height: groups.length ? Math.max(90, depth * 66 + 18) : 42,
                  }}
                >
                  {content.bars.map((bar) => (
                    <div
                      key={bar.bar}
                      className="reading-bar-line"
                      style={{ left: percent(bar.start) }}
                    />
                  ))}
                  {beat >= start && beat < end && (
                    <div
                      className="reading-cursor"
                      style={{ left: percent(beat) }}
                      aria-hidden="true"
                    />
                  )}
                  {rests.map((rest) => (
                    <div
                      className="reading-rest"
                      key={rest.beat}
                      style={{
                        left: percent(rest.beat),
                        width: widthPercent(
                          rest.beat,
                          rest.beat + rest.duration,
                        ),
                      }}
                      title={`Rest for ${readingDuration(rest.duration)} quarter-note beats`}
                    >
                      <span>rest</span>
                      <i />
                    </div>
                  ))}
                  {groups.map((group) => (
                    <div
                      className={`reading-chord ${group.held ? "reading-held" : ""}`}
                      key={`${group.beat}:${group.held}`}
                      style={{ left: percent(group.beat) }}
                    >
                      {group.notes.map((note, row) => {
                        const sounding =
                          beat >= note.beat - 0.000001 &&
                          beat < note.beat + note.duration - 0.000001;
                        const display = readingNoteDisplay(note, start);
                        const { held } = display;
                        const duration = readingDuration(display.duration);
                        return (
                          <button
                            type="button"
                            key={`${note.id || note.midi}:${note.beat}`}
                            className={`reading-note ${sounding ? "reading-current" : ""} ${held ? "reading-continuation" : ""}`}
                            style={{ top: row * 66 }}
                            onClick={() => seek(display.seekBeat)}
                            aria-label={`${noteName(note.midi)}, ${labels === "numbers" ? "number " : ""}${label(note.midi)}, ${held ? "remaining hold" : "hold"} ${duration} quarter-note beats${held ? ", continuing from previous passage" : ""}`}
                            aria-current={sounding ? "true" : undefined}
                            title={`${noteName(note.midi)} · ${duration} quarter-note beats${held ? " remaining · already held" : ""}`}
                          >
                            <strong>
                              {held && <i aria-hidden="true">↳</i>}
                              {label(note.midi)}
                            </strong>
                            <span>{noteName(note.midi)}</span>
                            <small>
                              ♩ {duration}
                              {held && <em>remaining</em>}
                            </small>
                          </button>
                        );
                      })}
                    </div>
                  ))}
                </div>
              </div>
            );
          })}
        </div>
      </div>
      <footer className="reading-footer">
        <span>
          {labels === "numbers"
            ? `1 = ${noteName(60 + tonic).replace(/\d+$/, "")} · Numbers are notes, not fingers`
            : labels === "solfege"
              ? `Do = ${noteName(60 + tonic).replace(/\d+$/, "")}`
              : "Letters name the keys"}
        </span>
        <span>
          C4 = middle C <i>·</i> ♩ = one quarter-note beat <i>·</i> spacing
          adapts
          {passage.detail ? " · detailed passage" : ""}
        </span>
      </footer>
    </section>
  );
}
